import java.util.*;import net.minecraft.item.*;import net.minecraft.util.text.TextFormatting;
public class RepresentativeItemTooltips {
public static class ItemGlyphBlock extends Item { @Override public void addInformation(ItemStack itemstack, net.minecraft.world.World tooltipWorld, List<String> list, net.minecraft.client.util.ITooltipFlag tooltipFlag) {
		
		switch(itemstack.getItemDamage()) {
		case 0: list.add("Hourglass"); break;
		case 1: list.add("Eye"); break;
		case 2: list.add("'Pillar'"); break;
		case 3: list.add("IOI"); break;
		case 4: list.add("Delta"); break;
		case 5: list.add("VTPC"); break;
		case 6: list.add("Cool S"); break;
		case 7: list.add("Trefoil"); break;
		case 8: list.add("Pony"); break;
		case 9: list.add("Sparkle"); break;
		case 10: list.add("PiP"); break;
		case 11: list.add("Triangles"); break;
		case 12: list.add("Linux Mint"); break;
		case 13: list.add("13"); break;
		case 14: list.add("Digamma"); break;
		case 15: list.add("Celestial Altar"); break;
		}
	}
}
public static class ItemMissileShuttle extends Item { @Override public void addInformation(ItemStack itemstack, net.minecraft.world.World tooltipWorld, List<String> list, net.minecraft.client.util.ITooltipFlag tooltipFlag) {
		list.add("Tonite, on bo''om gear:");
		list.add("James huffs leaded gasoline and");
		list.add("goes insane, Richard spends the");
		list.add("entire budget on a broken .PNG,");
		list.add("And I forget to set the infinite");
		list.add("Water tanks on our RBMK to flow");
		list.add("out, blowing up our entire base");
	}
}
public static class ItemTemFlakes extends Item { @Override public void addInformation(ItemStack itemstack, net.minecraft.world.World tooltipWorld, List<String> list, net.minecraft.client.util.ITooltipFlag tooltipFlag) {
		if(itemstack.getItemDamage() == 0)
		{
			list.add("Heals 2HP DISCOUNT FOOD OF TEM!!!");
		}
		if(itemstack.getItemDamage() == 1)
		{
			list.add("Heals 2HP food of tem");
		}
		if(itemstack.getItemDamage() == 2)
		{
			list.add("Heals food of tem (expensiv)");
		}
	}
}
public static class ItemReactorSensor extends Item { @Override public void addInformation(ItemStack itemstack, net.minecraft.world.World tooltipWorld, List<String> list, net.minecraft.client.util.ITooltipFlag tooltipFlag) {
		if (itemstack.getTagCompound() != null) {
			list.add("x: " + itemstack.getTagCompound().getInteger("x"));
			list.add("y: " + itemstack.getTagCompound().getInteger("y"));
			list.add("z: " + itemstack.getTagCompound().getInteger("z"));
		} else {
			list.add("No reactor selected!");
		}
	}
}
public static class ItemStarmetal extends Item { @Override public void addInformation(ItemStack stack, net.minecraft.world.World tooltipWorld, List<String> list, net.minecraft.client.util.ITooltipFlag tooltipFlag) {
		
		switch(stack.getItemDamage()) {
		case 1: list.add(TextFormatting.ITALIC + "Astra"); break;
		case 2: list.add(TextFormatting.ITALIC + "Ursa"); break;
		case 3: list.add(TextFormatting.ITALIC + "Orion"); break;
		}
	}
}
public static class ItemTeleLink extends Item { @Override public void addInformation(ItemStack itemstack, net.minecraft.world.World tooltipWorld, List<String> list, net.minecraft.client.util.ITooltipFlag tooltipFlag) {
		if (itemstack.getTagCompound() != null) {
			list.add("X: " + itemstack.getTagCompound().getInteger("x"));
			list.add("Y: " + itemstack.getTagCompound().getInteger("y"));
			list.add("Z: " + itemstack.getTagCompound().getInteger("z"));
			list.add("D: " + itemstack.getTagCompound().getInteger("dim"));
		} else {
			list.add(TextFormatting.RED + "Select exit location first!");
		}
	}
}
public static class ItemDetonator extends Item { @Override public void addInformation(ItemStack itemstack, net.minecraft.world.World tooltipWorld, List<String> list, net.minecraft.client.util.ITooltipFlag tooltipFlag) {
		list.add("Shift right-click to set position,");
		list.add("right-click to detonate!");
		if(itemstack.getTagCompound() == null) {
			list.add(TextFormatting.RED + "No position set!");
		} else {
			list.add(TextFormatting.YELLOW + "Linked to " + itemstack.getTagCompound().getInteger("x") + ", " + itemstack.getTagCompound().getInteger("y") + ", " + itemstack.getTagCompound().getInteger("z"));
		}
	}
}
public static class ItemWiring extends Item { @Override public void addInformation(ItemStack itemstack, net.minecraft.world.World tooltipWorld, List<String> list, net.minecraft.client.util.ITooltipFlag tooltipFlag) {
		if(itemstack.getTagCompound() != null) {
			list.add("Wire start x: " + itemstack.getTagCompound().getInteger("x"));
			list.add("Wire start y: " + itemstack.getTagCompound().getInteger("y"));
			list.add("Wire start z: " + itemstack.getTagCompound().getInteger("z"));
		} else {
			list.add("Right-click poles to connect");
		}
	}
}
}
